Recherches, Tris & Fichiers en Python

# Recherche dichotomique (T trié croissant) def rech_dicho(T, n, x): g, d = 0, n - 1 while g <= d: m = (g + d) // 2 if T[m] == x: return m elif T[m] < x: g = m + 1 else: d = m - 1 return -1
# Tri à bulles & Tri par sélection def tri_selection(T, n): for i in range(n - 1): pmin = i for j in range(i + 1, n): if T[j] < T[pmin]: pmin = j if pmin != i: T[i], T[pmin] = T[pmin], T[i]
