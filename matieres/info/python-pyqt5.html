Tableaux NumPy et enregistrements

En Bac Informatique, les tableaux statiques 1D, 2D et tableaux d'enregistrements sont implémentés avec numpy.array et les dictionnaires dict.

from numpy import array

# Tableau 1D de N entiers / chaînes / réels
T = array([int()] * N)
T_ch = array([str()] * N)      # ou dtype=object pour chaînes de longueur variable

# Matrice (Tableau 2D) de L lignes et C colonnes
M = array([[int()] * C] * L)   # Accès : M[i, j]

# Enregistrement (dictionnaire) et Tableau d'enregistrements
e = {"nom": "", "moy": 0.0}    # Accès : e["nom"]
TE = array([dict()] * N)       # Tableau de N enregistrements
for i in range(N):
    TE[i] = {"nom": input("Nom: "), "moy": float(input("Moy: "))}

Manipulation des fichiers en Python

Fichiers texte (.txt)

# Écriture dans un fichier texte

f.close()

# Lecture ligne par ligne jusqu'à la fin
f = open("donnees.txt", "r")
ligne = f.readline()
while ligne != "":
    ligne = ligne[:len(ligne) - 1] # ou ligne.strip() pour enlever "\n"
    # ... traitement de ligne ...
    ligne = f.readline()
f.close()

Fichiers typés / binaires (.dat avec pickle)

from pickle import dump, load

# Écriture d'enregistrements dans un fichier binaire
f = open("eleves.dat", "wb")       # "ab" pour ajouter en fin
dump(e, f)
f.close()

# Lecture de tous les enregistrements (boucle + try / except)
f = open("eleves.dat", "rb")
fin = False
while not fin:
    try:
        e = load(f)
        # ... traitement de e["nom"], e["moy"] ...
    except:
        fin = True
f.close()

Implémentation des tris, recherches et récursivité

# Recherche dichotomique (tableau T trié croissant)
def rech_dicho(T, n, x):
    g, d, trouve = 0, n - 1, False
    while g <= d and not trouve:
        m = (g + d) // 2
        if T[m] == x:
            trouve = True
        elif x < T[m]:
            d = m - 1
        else:
            g = m + 1
    return trouve

# Tri à bulles
def tri_bulles(T, n):
    echange = True
    while echange and n > 1:
        echange = False
        for i in range(n - 1):
            if T[i] > T[i + 1]:
                T[i], T[i + 1] = T[i + 1], T[i]
                echange = True
        n -= 1

# Tri par insertion
def tri_insertion(T, n):
    for i in range(1, n):
        tmp = T[i]
        j = i
        while j > 0 and T[j - 1] > tmp:
            T[j] = T[j - 1]
            j -= 1
        T[j] = tmp

# Exemple récursif : PGCD d'Euclide
def pgcd(a, b):
    if b == 0:
        return a
    return pgcd(b, a % b)

Interfaces graphiques avec PyQt5 (Qt Designer)

Squelette complet d'une application PyQt5

from PyQt5.uic import loadUi
from PyQt5.QtWidgets import QApplication, QMessageBox, QTableWidgetItem

def traiter():
    ch = fen.le_saisie.text()
    if ch == "":
        QMessageBox.critical(fen, "Erreur", "Veuillez remplir le champ !")
    else:
        fen.lbl_res.setText("Résultat : " + ch.upper())

app = QApplication([])
fen = loadUi("interface.ui")
fen.btn_valider.clicked.connect(traiter)
fen.show()
app.exec_()

Pièges classiques en Python / PyQt5

À éviter absolument le jour de la pratique
  1. Dans .clicked.connect(ma_fonction), ne mets jamais de parenthèses après le nom de la fonction.
  2. Dans un QTableWidget, QTableWidgetItem(...) attend toujours une chaîne de caractères : utilise str(...) pour les nombres.
  3. Pour un fichier binaire .dat, ouvre-le toujours en mode "rb", "wb" ou "ab" (avec le b).
